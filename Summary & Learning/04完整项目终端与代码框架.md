## 4. 通用脚本入口

~~~python
import argparse
from pathlib import Path

ROOT = Path(__file__).resolve().parents[1]


def parse_args():
    parser = argparse.ArgumentParser()
    parser.add_argument("--weights", type=Path, required=True)
    parser.add_argument("--data", type=Path, required=True)
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--device", default="0")
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--execute", action="store_true")
    return parser.parse_args()


def main():
    args = parse_args()
    args.output.mkdir(parents=True, exist_ok=False)
    check_inputs(args)
    write_status(args.output, "preflight_complete")

    if not args.execute:
        return 0

    run(args)
    write_status(args.output, "complete")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
~~~

## 9. L1 和 Taylor 重要性

L1：

~~~python
scores = (
    conv.weight.detach()
    .abs()
    .flatten(1)
    .mean(1)
)
indices = torch.argsort(scores)[:remove_count].cpu().tolist()
~~~

Taylor：

~~~python
model.train()

for module in model.modules():
    if isinstance(module, torch.nn.BatchNorm2d):
        module.eval()

modules = dict(model.named_modules())

for batch in loader:
    batch = {
        key: value.to(device)
        if isinstance(value, torch.Tensor) else value
        for key, value in batch.items()
    }
    batch["img"] = batch["img"].float() / 255.0

    model.zero_grad(set_to_none=True)
    loss, _ = model(batch)
    loss = loss.sum() / batch["img"].shape[0]

    if not torch.isfinite(loss):
        raise RuntimeError("non-finite loss")

    loss.backward()

    for name in candidate_layers:
        weight = modules[name].weight
        score = (
            weight.detach() * weight.grad.detach()
        ).abs().flatten(1).mean(1)
        totals[name].add_(score.double().cpu())
~~~

## 10. 结构化剪枝

单个候选：

~~~python
import copy
import torch_pruning as tp

trial = copy.deepcopy(model)
target_conv = dict(trial.named_modules())[layer_name]

graph = tp.DependencyGraph().build_dependency(
    trial, example_inputs=example,
)
group = graph.get_pruning_group(
    target_conv,
    tp.prune_conv_out_channels,
    idxs=indices,
)

if not graph.check_pruning_group(group):
    raise RuntimeError("invalid pruning group")

group.prune()

with torch.no_grad():
    trial(example)
~~~

贪心选择：

~~~python
best = None

for layer_name in candidate_layers:
    trial = copy.deepcopy(model)

    try:
        prune_one_step(trial, layer_name)
        _, candidate_gmacs = model_stats(trial, example)
        saved = current_gmacs - candidate_gmacs
        cost = candidate_cost(layer_name)
        objective = cost / saved

        candidate = {
            "model": trial,
            "gmacs": candidate_gmacs,
            "objective": objective,
        }

        if best is None or objective < best["objective"]:
            best = candidate
    except (RuntimeError, ValueError, KeyError, IndexError):
        del trial

if best is None:
    raise RuntimeError("no feasible pruning action")

model = best["model"]
current_gmacs = best["gmacs"]
~~~

~~~python
import torch.nn as nn
from ultralytics import YOLO


def shape_signature(model):
    return {
        name: (
            type(module).__name__,
            tuple(
                (key, tuple(value.shape))
                for key, value in module.state_dict().items()
            ),
        )
        for name, module in model.named_modules()
        if isinstance(module, (nn.Conv2d, nn.BatchNorm2d))
    }


expected = shape_signature(model)
reloaded = YOLO(str(output)).model

if shape_signature(reloaded) != expected:
    raise RuntimeError("structure mismatch")
~~~

## 12. BN 校准

~~~python
before = {
    key: value.detach().cpu().clone()
    for key, value in model.state_dict().items()
}

model.eval()

for parameter in model.parameters():
    parameter.requires_grad_(False)

for module in model.modules():
    if isinstance(module, torch.nn.BatchNorm2d):
        module.reset_running_stats()
        module.train()

with torch.inference_mode():
    for batch in loader:
        images = (
            batch["img"].to(device, non_blocking=True)
            .float() / 255.0
        )
        model(images)

model.eval()

for key, old in before.items():
    new = model.state_dict()[key].detach().cpu()

    if torch.equal(old, new):
        continue

    if not key.endswith((
        "running_mean",
        "running_var",
        "num_batches_tracked",
    )):
        raise RuntimeError(key)
~~~

## 14. 蒸馏

教师和对齐层：

~~~python
import torch.nn as nn
import torch.nn.functional as F

teacher = load_teacher().to(device).eval()

for parameter in teacher.parameters():
    parameter.requires_grad_(False)

projectors = nn.ModuleList(
    nn.Conv2d(student_c, teacher_c, 1, bias=False)
    for student_c, teacher_c
    in zip(student_widths, teacher_widths)
).to(device)
~~~

特征损失：

~~~python
with torch.no_grad():
    teacher_features = teacher(images)

level_losses = []

for student, teacher_feature, projector in zip(
    student_features, teacher_features, projectors,
):
    teacher_feature = F.interpolate(
        teacher_feature,
        size=student.shape[-2:],
        mode="bilinear",
        align_corners=False,
    )
    student = F.normalize(projector(student), dim=1)
    teacher_feature = F.normalize(teacher_feature, dim=1)

    level_losses.append(
        1.0 - (
            student * teacher_feature
        ).sum(dim=1).mean()
    )

kd_loss = sum(
    weight * loss
    for weight, loss in zip(level_weights, level_losses)
)
~~~

加入总损失：

~~~python
regular_loss, loss_items = super().loss(batch, preds)
~~~

~~~python
with torch.inference_mode():

    for _ in range(repeats):
        model(example)

    time.perf_counter() - started
~~~
